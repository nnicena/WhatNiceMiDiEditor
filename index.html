<script>

    let midiAccess = null;

    let selectedOutput = null;


    const statusDiv = document.getElementById('status');

    const outputSelect = document.getElementById('midi-outputs');

    const logBox = document.getElementById('log-box');


    function log(message) {

        logBox.innerHTML += `<br>>> ${message}`;

        logBox.scrollTop = logBox.scrollHeight;

    }


    // 1. ขอสิทธิ์เข้าถึง Web MIDI API จาก Browser

    if (navigator.requestMIDIAccess) {

        navigator.requestMIDIAccess({ sysex: true }) // เปิดใช้งาน sysex สำหรับโอนย้ายข้อมูล

            .then(onMIDISuccess, onMIDIFailure);

    } else {

        statusDiv.style.backgroundColor = "#ff4d4d";

        statusDiv.innerText = "เบราว์เซอร์นี้ไม่รองรับ Web MIDI API (แนะนำให้ใช้ Chrome หรือ Edge)";

    }


    function onMIDISuccess(midi) {

        midiAccess = midi;

        statusDiv.style.backgroundColor = "#28a745";

        statusDiv.innerText = "ระบบ MIDI พร้อมใช้งาน! กรุณาเลือกอุปกรณ์";

        log("เข้าถึง Web MIDI API สำเร็จ (เปิดโหมด SysEx แล้ว)");

        

        updateMIDIDevices();

        // ตรวจจับกรณีมีการดึงสายหรือเสียบสายอุปกรณ์ใหม่

        midiAccess.onstatechange = updateMIDIDevices;

    }


    function onMIDIFailure(error) {

        statusDiv.style.backgroundColor = "#ff4d4d";

        statusDiv.innerText = "ปฏิเสธการเข้าถึงระบบ MIDI: " + error;

    }


    // 2. อัปเดตรายชื่อพอร์ตอุปกรณ์ที่เชื่อมต่ออยู่เข้ามาใน Dropdown

    function updateMIDIDevices() {

        outputSelect.innerHTML = '<option value="">-- กรุณาเลือกอุปกรณ์ --</option>';

        const outputs = midiAccess.outputs;


        if(outputs.size === 0) {

            log("ไม่พบอุปกรณ์ MIDI เชื่อมต่ออยู่");

            return;

        }


        outputs.forEach((output) => {

            const option = document.createElement('option');

            option.value = output.id;

            option.innerText = output.name;

            outputSelect.appendChild(option);

        });

    }


    // ดักจับการเลือกอุปกรณ์

    outputSelect.addEventListener('change', (e) => {

        const id = e.target.value;

        if(id) {

            selectedOutput = midiAccess.outputs.get(id);

            log(`เลือกอุปกรณ์ส่งออก: ${selectedOutput.name}`);

        } else {

            selectedOutput = null;

        }

    });


    // 3. ส่งข้อมูล SysEx เมื่อกดปุ่ม Save

    document.getElementById('btn-save').addEventListener('click', () => {

        if (!selectedOutput) {

            alert('กรุณาเลือกอุปกรณ์ MIDI ก่อนครับ!');

            return;

        }


        // ดึงค่าต่างๆ จากหน้าเว็บ UI

        const switchNum = parseInt(document.getElementById('switch-select').value);

        const msgType = parseInt(document.getElementById('msg-type').value);

        const midiNum = parseInt(document.getElementById('midi-number').value);

        const midiVal = parseInt(document.getElementById('midi-value').value);


        /* 

           สร้างโครงสร้างชุดข้อความจำลองขึ้นมา (Byte Array SysEx)

           0xF0 = จุดเริ่มต้นข้อความ SysEx

           0x7D = รหัสผู้ผลิต (Manufacturer ID สำหรับงาน DIY / วิจัย)

           [ข้อมูลการตั้งค่าที่เราส่งไป]

           0xF7 = จุดสิ้นสุดข้อความ SysEx

        */

        const sysexData = [

            0xF0,       // Start SysEx

            0x7D,       // Educational/DIY ID

            switchNum,  // ปุ่มที่เท่าไหร่ (1, 2, 3)

            msgType,    // ประเภทข้อความ (176=CC, 192=PC)

            midiNum,    // เลขข้อความ (0-127)

            midiVal,    // ค่าความแรง (0-127)

            0xF7        // End SysEx

        ];


        try {

            // สั่งให้เว็บแอปพลิเคชันยิงข้อมูลดิบออกไปทางสาย USB ทันที

            selectedOutput.send(sysexData);

            log(`ส่งคำสั่งเซฟปุ่ม ${switchNum} สำเร็จ -> ชุดข้อมูลบิต: [${sysexData.map(x => '0x' + x.toString(16).toUpperCase()).join(', ')}]`);

            alert('ส่งข้อมูลโครงสร้างปุ่มไปยังฟุตสวิตช์เรียบร้อยแล้ว!');

        } catch (error) {

            log(`เกิดข้อผิดพลาดในการส่งข้อมูล: ${error}`);

            alert('ไม่สามารถส่งข้อมูลได้ กรุณาตรวจสอบการเชื่อมต่อ');

        }

    });

</script>
