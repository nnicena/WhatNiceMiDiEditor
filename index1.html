<script>



/* =========================

   DATA

========================= */


let banks = [];


let currentBank = 0;


let currentSwitch = 0;


let clipboard = null;


let serialPort = null;



/* =========================

   CREATE SWITCH

========================= */


function createSwitch(index) {


    const names = [

        "DIST",

        "DELAY",

        "REVERB",

        "SOLO",

        "CHORUS",

        "TAP"

    ];


    return {


        shortName: names[index],


        longName: names[index],


        column: [0,8,16,0,8,16][index],


        row: index < 3 ? 0 : 1,


        action: "Press",


        midiType: "CC",


        channel: 1,


        number: 20 + index,


        value: 127


    };


}



/* =========================

   CREATE BANK

========================= */


function createBank(name) {


    return {


        name: name,


        switches: [


            createSwitch(0),

            createSwitch(1),

            createSwitch(2),

            createSwitch(3),

            createSwitch(4),

            createSwitch(5)


        ]


    };


}



/* =========================

   INITIAL DATA

========================= */


banks.push(

    createBank("BANK 01")

);


banks.push(

    createBank("BANK 02")

);



/* =========================

   RENDER

========================= */


function render() {


    renderBanks();


    renderSwitches();


    loadEditor();


    renderLCD();


}



/* =========================

   BANKS

========================= */


function renderBanks() {


    const container =

        document.getElementById("banks");


    container.innerHTML = "";


    banks.forEach((bank,index)=>{


        const button =

            document.createElement("button");


        button.className =

            "bank-button " +

            (index === currentBank

                ? "active"

                : "");


        button.textContent =

            bank.name;


        button.onclick = ()=>{


            currentBank = index;


            currentSwitch = 0;


            render();


        };


        container.appendChild(button);


    });


}



/* =========================

   SWITCHES

========================= */


function renderSwitches() {


    const container =

        document.getElementById("switches");


    container.innerHTML = "";


    const switches =

        banks[currentBank].switches;


    switches.forEach((sw,index)=>{


        const button =

            document.createElement("button");


        button.className =

            "switch " +

            (index === currentSwitch

                ? "active"

                : "");


        button.innerHTML = `


            <div>FS${index+1}</div>


            <div class="switch-name">

                ${sw.shortName}

            </div>


            <small>

                ${sw.midiType}

                ${sw.number}

            </small>


        `;


        button.onclick = ()=>{


            currentSwitch = index;


            render();


        };


        container.appendChild(button);


    });


}



/* =========================

   LOAD EDITOR

========================= */


function loadEditor() {


    const sw =

        banks[currentBank]

        .switches[currentSwitch];



    document.getElementById(

        "selectedSwitch"

    ).textContent =

        "FS" + (currentSwitch + 1);



    document.getElementById(

        "shortName"

    ).value =

        sw.shortName;



    document.getElementById(

        "longName"

    ).value =

        sw.longName;



    document.getElementById(

        "lcdColumn"

    ).value =

        sw.column;



    document.getElementById(

        "lcdRow"

    ).value =

        sw.row;



    document.getElementById(

        "action"

    ).value =

        sw.action;



    document.getElementById(

        "midiType"

    ).value =

        sw.midiType;



    document.getElementById(

        "channel"

    ).value =

        sw.channel;



    document.getElementById(

        "number"

    ).value =

        sw.number;



    document.getElementById(

        "value"

    ).value =

        sw.value;


}



/* =========================

   APPLY

========================= */


function applySettings() {


    const sw =

        banks[currentBank]

        .switches[currentSwitch];



    sw.shortName =

        document.getElementById(

            "shortName"

        ).value;



    sw.longName =

        document.getElementById(

            "longName"

        ).value;



    sw.column =

        Number(

            document.getElementById(

                "lcdColumn"

            ).value

        );



    sw.row =

        Number(

            document.getElementById(

                "lcdRow"

            ).value

        );



    sw.action =

        document.getElementById(

            "action"

        ).value;



    sw.midiType =

        document.getElementById(

            "midiType"

        ).value;



    sw.channel =

        Number(

            document.getElementById(

                "channel"

            ).value

        );



    sw.number =

        Number(

            document.getElementById(

                "number"

            ).value

        );



    sw.value =

        Number(

            document.getElementById(

                "value"

            ).value

        );



    render();


    setStatus(

        "Settings applied"

    );


}



/* =========================

   LCD

========================= */


function renderLCD() {


    let row1 =

        "                        ";


    let row2 =

        "                        ";



    banks[currentBank]

    .switches

    .forEach(sw=>{


        let text =

            sw.shortName.substring(

                0,8

            );



        let column =

            Math.max(

                0,

                Math.min(

                    23 - text.length,

                    sw.column

                )

            );



        if(sw.row === 0) {


            row1 =

                row1.substring(

                    0,column

                )

                +

                text

                +

                row1.substring(

                    column + text.length

                );


        } else {


            row2 =

                row2.substring(

                    0,column

                )

                +

                text

                +

                row2.substring(

                    column + text.length

                );


        }


    });



    document.getElementById(

        "lcd"

    ).textContent =

        row1 + "\n" + row2;


}



/* =========================

   ADD BANK

========================= */


function addBank() {


    const number =

        banks.length + 1;


    banks.push(

        createBank(

            "BANK " +

            String(number)

            .padStart(2,"0")

        )

    );


    currentBank =

        banks.length - 1;


    currentSwitch = 0;


    render();


}



/* =========================

   COPY

========================= */


function copySwitch() {


    clipboard =

        JSON.parse(

            JSON.stringify(

                banks[currentBank]

                .switches[currentSwitch]

            )

        );


    setStatus(

        "FS copied"

    );


}



/* =========================

   PASTE

========================= */


function pasteSwitch() {


    if(!clipboard)

        return;



    banks[currentBank]

    .switches[currentSwitch] =

        JSON.parse(

            JSON.stringify(

                clipboard

            )

        );



    render();


    setStatus(

        "FS pasted"

    );


}



/* =========================

   SAVE

========================= */


function saveSettings() {


    localStorage.setItem(

        "niceMidiSettings",

        JSON.stringify(banks)

    );


    setStatus(

        "Saved to browser"

    );


}



/* =========================

   EXPORT

========================= */


function exportJSON() {


    const data =

        JSON.stringify(

            banks,

            null,

            2

        );



    const blob =

        new Blob(

            [data],

            {

                type:

                "application/json"

            }

        );



    const url =

        URL.createObjectURL(blob);



    const a =

        document.createElement("a");



    a.href = url;


    a.download =

        "nice-midi-settings.json";


    a.click();


    URL.revokeObjectURL(url);


}



/* =========================

   IMPORT

========================= */


function importJSON(event) {


    const file =

        event.target.files[0];



    const reader =

        new FileReader();



    reader.onload =

        function(){


            try {


                banks =

                    JSON.parse(

                        reader.result

                    );


                currentBank = 0;


                currentSwitch = 0;


                render();


                setStatus(

                    "Import successful"

                );


            }


            catch {


                setStatus(

                    "Invalid JSON"

                );


            }


        };



    reader.readAsText(file);


}



/* =========================

   ESP32 CONNECT

========================= */


async function connectESP32() {


    if(!("serial" in navigator)) {


        setStatus(

            "Web Serial ไม่รองรับใน browser นี้"

        );


        return;


    }



    try {


        serialPort =

            await navigator.serial

            .requestPort();



        await serialPort.open({

            baudRate: 115200

        });



        setStatus(

            "ESP32 Connected"

        );


    }


    catch(error) {


        console.error(error);


        setStatus(

            "ESP32 connection failed"

        );


    }


}



/* =========================

   SEND TO ESP32

========================= */


async function sendToESP32() {


    if(!serialPort) {


        setStatus(

            "Connect ESP32 ก่อน"

        );


        return;


    }



    const data =

        JSON.stringify(banks);



    const writer =

        serialPort.writable

        .getWriter();



    const encoder =

        new TextEncoder();



    await writer.write(

        encoder.encode(

            data + "\n"

        )

    );



    writer.releaseLock();



    setStatus(

        "Settings sent to ESP32"

    );


}



/* =========================

   READ ESP32

========================= */


async function readFromESP32() {


    if(!serialPort) {


        setStatus(

            "Connect ESP32 ก่อน"

        );


        return;


    }



    const reader =

        serialPort.readable

        .getReader();



    try {


        const {

            value,

            done

        } =

        await reader.read();



        if(done)

            return;



        const text =

            new TextDecoder()

            .decode(value);



        console.log(

            "ESP32:",

            text

        );



        try {


            banks =

                JSON.parse(text);


            currentBank = 0;


            currentSwitch = 0;


            render();


            setStatus(

                "Settings received"

            );


        }


        catch {


            setStatus(

                "Received data is not JSON"

            );


        }


    }


    finally {


        reader.releaseLock();


    }


}



/* =========================

   STATUS

========================= */


function setStatus(text) {


    document.getElementById(

        "status"

    ).textContent =

        text;


}



/* =========================

   LOAD SAVED DATA

const saved =

    localStorage.getItem(

        "niceMidiSettings"

    );



if(saved) {


    try {


        banks =

            JSON.parse(saved);


    }


    catch {}


}



/* =========================

   START

========================= */


render();
